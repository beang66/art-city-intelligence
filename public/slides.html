export function proxy(request: NextRequest) {
  const user = process.env.ADMIN_USER
  const password = process.env.ADMIN_PASSWORD
  const expected = user && password
    ? `Basic ${btoa(`${user}:${password}`)}` : null
  // Fails closed: no login configured → locked
  if (expected &&
      request.headers.get('authorization') === expected)
    return NextResponse.next()
  return new NextResponse('Authentication required',
    { status: 401, headers: { 'WWW-Authenticate': … } })
}
export const config = { matcher: '/admin/:path*' }

6개 차원이 도시마다 같은 모양입니다.
1 · 분석

차원 점수 = 지수 × 고정 가중치

모든 도시에서 실험 > 시장 > 영향력 순서가 같다. '순위 하나는 강점의 조합을 숨긴다'는 기획 질문에 아직 답하지 못한다.

2 · 재현성

데이터가 코드 안에 손으로 입력됨

관측 순위를 PDF에서 옮겨 적었고 수집일·원자료 파일이 없다. 값을 다시 만들거나 대조할 방법이 없다.

3 · 권리

공개 저장소에 보고서 PDF 13개 (약 100MB)

단행본 PDF와 작품 이미지 20점이 함께 올라가 있다. 인용은 링크로, 이미지는 이용 허락 확인이 필요하다.

4 · 테스트

자동 테스트 없음

중앙값·순위 계산, API 상태 코드, 인증이 사람의 확인에만 기대고 있다.
